theorem VIR.slice_fuse_dep : ∀ {n : Nat} (b : Slice n) (i j : Nat) (h₁ : i ≤ n) (h₂ : j ≤ n - i) (h₃ : i + j ≤ n)
  (e : (m : Nat) → m = n - (i + j) → Slice m → Out),
  (advance b i h₁ fun s₁ => advance s₁ j h₂ fun s₂ => e (n - i - j) ⋯ s₂) =
    advance b (i + j) h₃ fun s₂ => e (n - (i + j)) ⋯ s₂ :=
fun {n} b i j h₁ h₂ h₃ e =>
  id
    (Slice.congr_heq_dep e (Decidable.byContradiction fun a => slice_fuse_dep._proof_1 i j a) rfl
      (Slice.adv_adv_heq b i j h₁ h₂ h₃))
theorem VIR.Slice.congr_heq_dep : ∀ {N : Nat} (e : (m : Nat) → m = N → Slice m → Out) {m₁ m₂ : Nat} (h₁ : m₁ = N)
  (h₂ : m₂ = N) {x : Slice m₁} {y : Slice m₂}, x ≍ y → e m₁ h₁ x = e m₂ h₂ y :=
fun {N} e {m₁ m₂} h₁ h₂ {x} {y} hxy =>
  Eq.rec (motive := fun {N} h₁ => ∀ (e : (m : Nat) → m = N → Slice m → Out) (h₂ : m₂ = N), e m₁ h₁ x = e m₂ h₂ y)
    (fun e h₂ =>
      Eq.rec (motive := fun {m₁} h₂ =>
        ∀ {x : Slice m₁}, x ≍ y → ∀ (e : (m : Nat) → m = m₁ → Slice m → Out), e m₁ (Eq.refl m₁) x = e m₂ h₂ y)
        (fun {x} hxy e =>

            Slice m₂ = β → y ≍ a → hxy ≍ t → e m₂ (Eq.refl m₂) x = e m₂ (Eq.refl m₂) y) hxy
